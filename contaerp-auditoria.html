📋 Resumen del Proyecto

ConTaERP — Sistema de contabilidad y ERP para Guatemala (GLL Consulting).
Multi-tenant: Cada empresa tiene su propio schema en PostgreSQL. El TenantDataSourceService mantiene un pool de DataSource por tenant.
Auth: JWT (accessToken + refreshToken). Header obligatorio: X-Tenant-ID.
Guards (orden obligatorio): JwtAuthGuard → TenantGuard → PermissionsGuard.
Numeración: SequenceService centralizado. Series: FAC · ANT · NC · PAG · POL · COT.
Contabilidad: JournalService genera pólizas automáticas (type=AUTO) en facturas, pagos y notas de crédito.
FEL Guatemala: Tipos FACT/FCAM/FPEQ/FAPE/NABN/RECI/NDEB/NCRE. Frases fiscales: Art.52, Art.55, Exportación, Art.7.
Anticipos: InvoiceType.ADVANCE — excluidos de GET /ventas/facturas con WHERE type != 'advance'.

TS Generador centralizado de números correlativos — reemplaza 6+ implementaciones duplicadas
@Injectable()
export class SequenceService {
  async nextNumber(
    ds: DataSource,
    series: string,              // 'FAC' | 'ANT' | 'NC' | 'PAG' | 'POL' | 'COT'
    entityClass: new () => any,
    numberField: string,
    opts: { withDeleted?: boolean; padLength?: number; yearSuffix?: boolean } = {},
  ): Promise<string> {
    const { withDeleted = true, padLength = 5, yearSuffix = true } = opts;
    const year   = new Date().getFullYear();
    const prefix = yearSuffix ? `${series}-${year}-` : `${series}-`;

    const qb = ds.getRepository(entityClass)
      .createQueryBuilder('e')
      .where(`e.${numberField} LIKE :p`, { p: `${prefix}%` })
      .orderBy(`e.${numberField}`, 'DESC')
      .limit(1);

    if (withDeleted) qb.withDeleted(); // evita reusar números de registros borrados

    const last = await qb.getOne().catch(() => null);
    let seq = 1;
    if (last) {
      const m = (last[numberField] as string).match(/(\d+)$/);
      if (m) seq = parseInt(m[1], 10) + 1;
    }
    return `${prefix}${String(seq).padStart(padLength, '0')}`;
  }
}

@Injectable()
export class JournalService {
  // Resuelve cuenta por ID, con fallback por código en el catálogo
  async resolveAccount(ds, accountId?, fallbackCode?, fallbackName?): Promise<ResolvedAccount> {
    if (accountId) {
      const acc = await accRepo.findOne({ where: { id: accountId } });
      if (acc) return { id: acc.id, code: acc.code, name: acc.name };
    }
    if (fallbackCode) {
      const acc = await accRepo.findOne({ where: { code: fallbackCode } });
      if (acc) return { id: acc.id, code: acc.code, name: acc.name };
    }
    return { id: undefined, code: fallbackCode ?? '????', name: fallbackName ?? 'Cuenta no configurada' };
  }

  // Crea póliza completa con número POL-YYYY-NNNNN correlativo
  // Solo inserta líneas con accountId válido (respeta FK de BD)
  async postEntry(ds: DataSource, input: PostEntryInput): Promise<string> {
    const entryNumber = await this.sequenceService.nextNumber(
      ds, 'POL', JournalEntryEntity, 'entryNumber', { withDeleted: false }
    );
    const entry = await jeRepo.save(jeRepo.create({
      entryNumber, type: input.type ?? JournalEntryType.AUTO,
      status: input.status ?? JournalEntryStatus.POSTED,
      ...input
    }));
    const validLines = input.lines.filter(l => l.accountId);
    for (const l of validLines) {
      await jelRepo.save(jelRepo.create({ journalEntryId: entry.id, ...l }));
    }
    return entry.id;
  }

  // Elimina póliza (CASCADE elimina líneas automáticamente)
  async deleteEntry(ds: DataSource, journalEntryId: string): Promise<void> {
    await ds.getRepository(JournalEntryEntity).delete(journalEntryId).catch(() => null);
  }
}

TSX LineItem · recalc() · CellInput con debounce · IVA Guatemala incluido
export interface LineItem {
  _key: string; productId?: string; description: string;
  unit?: string; quantity: number; unitPrice: number;
  discountPercent: number; taxPercent: number;
  taxInclusive?: boolean;  // true por defecto (Guatemala: precio incluye IVA)
  accountId?: string; projectId?: string;
  lineNet: number; lineTax: number; lineTotal: number;
}

export function recalc(item: LineItem): LineItem {
  const gross = item.quantity * item.unitPrice * (1 - item.discountPercent / 100);
  if (item.taxInclusive && item.taxPercent > 0) {
    // Precio YA incluye IVA → extrae base e impuesto
    lineNet   = Math.round(gross / (1 + item.taxPercent / 100) * 100) / 100
    lineTax   = Math.round((gross - lineNet) * 100) / 100
    lineTotal = Math.round(gross * 100) / 100
  } else {
    // Precio excluye IVA → agrega impuesto sobre base
    lineNet   = Math.round(gross * 100) / 100
    lineTax   = Math.round(lineNet * item.taxPercent / 100 * 100) / 100
    lineTotal = lineNet + lineTax
  }
}

// CellInput — evita pérdida de foco en Ant Design Table
// Problema: Table re-renderiza en cada onChange → pierde el cursor
// Solución: estado local + debounce 500ms + commit en onBlur/Enter
const CellInput = memo(({ value, onCommit, placeholder, style }) => {
  const [local, setLocal] = useState(value)
  const committed = useRef(value)
  const debounceRef = useRef()

  useEffect(() => {
    if (value !== committed.current) { setLocal(value); committed.current = value }
  }, [value])

  const handleChange = (e) => {
    setLocal(e.target.value)
    clearTimeout(debounceRef.current)
    debounceRef.current = setTimeout(() => {
      if (e.target.value !== committed.current) {
        committed.current = e.target.value
        onCommit(e.target.value)
      }
    }, 500)
  }
  return <Input value={local} onChange={handleChange} onBlur={() => commitNow(local)} />
})
